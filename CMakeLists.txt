cmake_minimum_required(VERSION 3.20)
project(dome-torrent VERSION 0.1.0 LANGUAGES C CXX)
if(POLICY CMP0167)
  cmake_policy(SET CMP0167 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(DOME_TORRENT_ENABLE_X11 "Enable explicit X11 appmenu association" OFF)
find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
find_package(LibtorrentRasterbar 2.0 CONFIG REQUIRED)
find_package(OpenGL REQUIRED)
find_package(Threads REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig zlib)
add_library(torrent-core native/engine.cpp)
target_include_directories(torrent-core PUBLIC native)
target_link_libraries(torrent-core PUBLIC LibtorrentRasterbar::torrent-rasterbar Threads::Threads)
target_compile_options(torrent-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(dome-torrent native/main.cpp)
target_link_libraries(dome-torrent PRIVATE torrent-core fltk::fltk-shared fltk::gl-shared fltk::images-shared OpenGL::GL PkgConfig::FONTCONFIG nlohmann_json::nlohmann_json)
target_compile_options(dome-torrent PRIVATE -Wall -Wextra -Wpedantic)
set(XNEDIT_ENABLE_X11 ${DOME_TORRENT_ENABLE_X11})
target_compile_definitions(dome-torrent PRIVATE XNEDIT_ENABLE_X11=$<BOOL:${DOME_TORRENT_ENABLE_X11}>)
include(native/desktop-menu.cmake)
de_xr_desktop_menu(dome-torrent)
set_target_properties(dome-torrent PROPERTIES INSTALL_RPATH "/opt/fltk-vr/toolkit/lib")
install(TARGETS dome-torrent RUNTIME DESTINATION bin)
# Route through DomeD before GApplication forwards to an existing instance.
# A plain launcher remains usable on machines without the optional helper.
find_program(DOME_TORRENT_LINK_HELPER NAMES dome-link)
set(DOME_TORRENT_DESKTOP_EXEC "dome-torrent")
if(DOME_TORRENT_LINK_HELPER)
  set(DOME_TORRENT_DESKTOP_EXEC "\"${DOME_TORRENT_LINK_HELPER}\" --app-id org.domed.Torrent -- dome-torrent")
endif()
configure_file(packaging/org.domed.Torrent.desktop.in org.domed.Torrent.desktop @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/org.domed.Torrent.desktop" DESTINATION share/applications)
install(FILES packaging/org.domed.Torrent.svg DESTINATION share/icons/hicolor/scalable/apps)
foreach(size 16 24 32 48 64 128 256)
  install(FILES "packaging/icons/${size}x${size}/org.domed.Torrent.png"
    DESTINATION "share/icons/hicolor/${size}x${size}/apps")
endforeach()
install(FILES README.md LICENSE DESTINATION share/doc/dome-torrent)
include(CTest)
if(BUILD_TESTING)
  add_executable(torrent-tests tests/engine.cpp)
  target_link_libraries(torrent-tests PRIVATE torrent-core)
  target_compile_options(torrent-tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  add_test(NAME magnet-transfer-and-resume COMMAND torrent-tests)
  set_tests_properties(magnet-transfer-and-resume PROPERTIES TIMEOUT 100)
endif()
